## 메타데이터

> [!info]+ YAML (표준 키 + 한국어 주석)
> ```yaml
> meta_level: "L1"                  # (메타레벨) 제1기-간략
> title: "<% tp.file.title %>"      # (사건명)
> 
> when: ""                          # (시점) 예: "1172~1356년" / "(추정) 기원전 72년"
> where: ""                         # (위치) 예: "셰닐리츠 왕국 (구체 지리 위치 미상)"
> where_roman: ""                   # (위치-로마자) 예: "xhenilrits"
> 
> summary: ""                       # (요약) 1~2문장
> 
> domains: []                       # (사건유형-1단계, 복수) 예: ["재난","자연·환경","인구·이주·개척"]
> actors:                           # (행위자, 간략)
>   - type: "국가"                  # (유형) 국가/민족·부족/종교단체/기업체/조직/개인/연합체/가문
>     name: ""                      # (명칭) 예: "셰닐리츠 왕국"
>     roman: ""                     # (로마자) 예: "xhenilrits"
>     role: ["주도"]                # (역할) 예: ["주도"], ["피해"], ["가해"], ["중재"]
> 
> reliability:                      # (신뢰성)
>   level_m2_p2: 0                  # (-2~+2)
>   basis:                          # (근거; 2단계)
>     - tier1: ""                   # 예: "전승" / "선전" / "학술" / "추정" / "미상" / "공식" / "정보"
>       tier2: ""                   # 예: "구전·민담" / "검열·편집" / "2차연구" / "재구성"
>   notes: ""                       # (노트) 왜 이 레벨인지 1~2문장
> 
> index_terms: []                   # (검색키워드) 약칭/별칭/오탈자/다언어 표기 등
> ```
